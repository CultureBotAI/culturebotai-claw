PREFIX=Path('scripts/adhoc/cycle-maintenance-20261115/provenance-v3')
def sha(raw):return hashlib.sha256(raw).hexdigest()
def main():
 p=argparse.ArgumentParser(description=__doc__);p.add_argument('--work-area',type=Path,required=True);p.add_argument('--staging',type=Path,required=True);a=p.parse_args()
 payload={};provenance=[]
 for source,(dest,expected) in PINS.items():
  path=a.work_area/source
  assert path.is_file() and not path.is_symlink()
  raw=path.read_bytes();assert sha(raw)==expected

  assert not (a.staging/path).exists()
