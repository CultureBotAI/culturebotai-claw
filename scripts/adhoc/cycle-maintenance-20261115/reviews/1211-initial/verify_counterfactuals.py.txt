CHILD = '''import importlib,json,pathlib,sys,pytest
case=json.loads(sys.argv[1]);root=pathlib.Path.cwd()
for sub in ('tests','src','.'):
    sys.path.insert(0,str(root/sub))
if case.get('mutation'):
    module=importlib.import_module(case['module'])
    source=(root/case['source']).read_text()
    assert source.count(case['old']) == 1
    exec(compile(source.replace(case['old'],case['new']),module.__file__,'exec'),module.__dict__)
sys.exit(pytest.main(['-q',case['test']]))
'''

def digest(raw):
    return hashlib.sha256(raw).hexdigest()

def main():
    sources={c['source']:digest((ROOT/c['source']).read_bytes()) for c in CASES}
    results=[]
    for case in CASES:
        for mutation in (False,True):
            with tempfile.TemporaryDirectory(prefix='pr1211-review-cache-') as cache:
                env=dict(os.environ,PYTHONPYCACHEPREFIX=cache)
                proc=subprocess.run([sys.executable,'-c',CHILD,json.dumps(dict(case,mutation=mutation))],
                    cwd=ROOT,env=env,text=True,capture_output=True,timeout=90)
            expected=1 if mutation else 0
            passed=proc.returncode==expected and ('1 failed' if mutation else '1 passed') in proc.stdout
            results.append({'name':case['name'],'issue':case['issue'],'mutation':mutation,
                'exit_code':proc.returncode,'expected_exit':expected,'passed':passed,
                'stdout_sha256':digest(proc.stdout.encode()),'stderr_sha256':digest(proc.stderr.encode())})
            if not passed:
                print(proc.stdout,file=sys.stderr);print(proc.stderr,file=sys.stderr)
    unchanged=sources=={p:digest((ROOT/p).read_bytes()) for p in sources}
    print(json.dumps({'sources':sources,'results':results,'sources_unchanged':unchanged,
        'passed':unchanged and all(r['passed'] for r in results),
        'helper_sha256':digest(Path(__file__).read_bytes())},indent=2))
    return 0 if unchanged and all(r['passed'] for r in results) else 1

if __name__=='__main__':
    sys.exit(main())
